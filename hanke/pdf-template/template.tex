  {\raggedright\headingfont\bfseries\fontsize{23}{27}\selectfont\color{inknavy}}

  {\clearpage\headingfont\bfseries\fontsize{18}{22}\selectfont\color{inknavy}}

  {\headingfont\bfseries\fontsize{12.5}{15}\selectfont\color{accent}}
  {}{0pt}{}

\titleformat{\paragraph}[block]
  {\headingfont\bfseries\fontsize{11}{13}\selectfont\color{inknavy}}
